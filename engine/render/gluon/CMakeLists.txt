# engine/render/gluon — cd::gluon (phase1133, SL-C step 2)
# SOTA shader MODULE library (ADR-20260612-shader-library-architecture):
# the authoritative GLSL modules live in shaders/modules/*.glsl and are
# embedded into a generated header at configure time so the library is
# self-contained (standalone-product requirement).

set(CD_GLUON_MODULES
  math_common
  brdf
  brdf_w8aq         # phase1135: verbatim W8-AQ block (transitional)
  cotangent_frame   # phase1138: verbatim from prim.frag (Mikkelsen 2010 TBN)
  light_atten       # phase1138: verbatim from prim.frag (Frostbite windowed)
  ltc_polygon       # phase1138: verbatim from prim.frag (atan2 edge integral)
  tonemap
)

set(_cd_gluon_gen_dir "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(_cd_gluon_header "${_cd_gluon_gen_dir}/gluon_embedded.hpp")

set(_cd_gluon_embed "// GENERATED by engine/render/gluon/CMakeLists.txt — DO NOT EDIT.\n")
string(APPEND _cd_gluon_embed "// Source of truth: engine/render/gluon/shaders/modules/*.glsl\n")
string(APPEND _cd_gluon_embed "#pragma once\n\n")
string(APPEND _cd_gluon_embed "#include <cd/gluon/ModuleRegistry.hpp>\n\n")
string(APPEND _cd_gluon_embed "#include <array>\n\n")
string(APPEND _cd_gluon_embed "namespace cd::gluon::detail\n{\n\n")

foreach(_mod IN LISTS CD_GLUON_MODULES)
  set(_mod_path "${CMAKE_CURRENT_SOURCE_DIR}/shaders/modules/${_mod}.glsl")
  if(NOT EXISTS "${_mod_path}")
    message(FATAL_ERROR "[cd::gluon] missing module: ${_mod_path}")
  endif()
  # Editing a module must re-run configure so the embed regenerates.
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_mod_path}")
  file(READ "${_mod_path}" _mod_content)
  if(_mod_content MATCHES "\\)CDSL\"")
    message(FATAL_ERROR "[cd::gluon] ${_mod}.glsl contains the raw-literal delimiter )CDSL\"")
  endif()
  string(APPEND _cd_gluon_embed "inline constexpr char kModule_${_mod}[] = R\"CDSL(${_mod_content})CDSL\";\n\n")
endforeach()

# The catalogue table is SORTED by virtual_path (ModuleRegistry.hpp
# contract); the authoring list above stays in dependency order.
set(_cd_gluon_sorted ${CD_GLUON_MODULES})
list(SORT _cd_gluon_sorted)
string(APPEND _cd_gluon_embed "inline constexpr std::array<ModuleDesc, ")
list(LENGTH CD_GLUON_MODULES _cd_gluon_count)
string(APPEND _cd_gluon_embed "${_cd_gluon_count}> kEmbeddedModules { {\n")
foreach(_mod IN LISTS _cd_gluon_sorted)
  string(APPEND _cd_gluon_embed "    ModuleDesc { \"cd/gluon/${_mod}.glsl\", kModule_${_mod} },\n")
endforeach()
string(APPEND _cd_gluon_embed "} };\n\n}  // namespace cd::gluon::detail\n")

# Only rewrite when the content actually changed (keeps mtime stable so
# no-op re-configures do not trigger rebuilds).
set(_cd_gluon_old "")
if(EXISTS "${_cd_gluon_header}")
  file(READ "${_cd_gluon_header}" _cd_gluon_old)
endif()
if(NOT _cd_gluon_old STREQUAL _cd_gluon_embed)
  file(WRITE "${_cd_gluon_header}" "${_cd_gluon_embed}")
endif()

cd_add_library(gluon
  TYPE STATIC
  EXCLUDE_FROM_INSTALL  # transitive on cd::shader (vendored glslang)
  SOURCES
    src/ModuleRegistry.cpp
  PUBLIC_DEPS
    cd::core
    cd::shader
)
target_include_directories(cd_gluon PRIVATE "${_cd_gluon_gen_dir}")

if(CD_ENABLE_TESTING)
  add_subdirectory(tests)
endif()
